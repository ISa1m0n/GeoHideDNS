# Фиксация проксируемых доменов на одном живом EU-сервере GeoHide.
# Правила:
#  - Формат: домен IP. Закрепление действует на домен и все его поддомены,
#    остаётся ТОЛЬКО указанный IP.
#  - IP должен присутствовать в исходных hosts (EU), из которых идёт сборка.
#    Если указать адрес, которого там нет, домен выпадет из hosts-ai.txt.
#    Домены с US-адресом (Gemini) здесь не закрепляются, они в custom-hosts.txt (us-domains.txt для адресов вне US-hosts не годится).
#  - Порядок важен: сборка берёт ПЕРВУЮ подходящую строку, поэтому более
#    конкретный домен (www.*) должен стоять ВЫШЕ общего (apex).
#  - Адреса задаёт сборка: @MAIN и @ALT берутся из health.txt (пишет tools/geohide-probe.ps1) или из defaults.txt.
#    @MAIN: основной живой EU-адрес. @ALT: второй, только для apex в разделе "Разнос www и apex".

openai.com                           @MAIN
oaistatic.com                        @MAIN

# Разнос www и apex по разным IP.
# Если оба имени на одном IP, Firefox склеивает их в одно HTTP/2-соединение,
# и сайт отвечает ошибкой (403 / Fastly unknown domain).
# Строка www.* стоит выше apex, иначе она не сработает.
www.last.fm                          @MAIN
last.fm                              @ALT
www.deezer.com                       @MAIN
deezer.com                           @ALT
